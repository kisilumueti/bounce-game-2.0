cmake_minimum_required(VERSION 3.20)
project(NeonBounce VERSION 2.0.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(bounce_engine STATIC engine/src/engine.cpp engine/src/random.c)
target_include_directories(bounce_engine PUBLIC engine/include)
target_compile_options(bounce_engine PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>)
add_executable(bounce_cli engine/src/main.cpp)
target_link_libraries(bounce_cli PRIVATE bounce_engine)
include(CTest)
if(BUILD_TESTING)
 add_executable(bounce_engine_test engine/tests/engine_test.cpp)
 target_link_libraries(bounce_engine_test PRIVATE bounce_engine)
 add_test(NAME engine_regression COMMAND bounce_engine_test)
endif()
if(EMSCRIPTEN)
 add_executable(bounce_wasm engine/src/engine.cpp engine/src/c_api.cpp)
 target_include_directories(bounce_wasm PRIVATE engine/include)
 set_target_properties(bounce_wasm PROPERTIES OUTPUT_NAME "bounce-core" SUFFIX ".js")
 target_link_options(bounce_wasm PRIVATE "-sEXPORTED_FUNCTIONS=['_bounce_reset','_bounce_jump','_bounce_step','_bounce_x','_bounce_y','_bounce_velocity_y','_bounce_score','_bounce_grounded','_bounce_jumps']" "-sEXPORTED_RUNTIME_METHODS=['ccall','cwrap']" "-sMODULARIZE=1" "-sEXPORT_NAME=createBounceCore" "-sALLOW_MEMORY_GROWTH=1" "-sENVIRONMENT=web,worker" "-sWASM=1")
endif()
